(function initCardReader(){
  let cardBuffer = '';
  let lastKeyTime = 0;
  const READER_TIMEOUT = 100; // ms between keypresses from card reader
  const CARD_MIN_LENGTH = 4;  // minimum card number length
  
  document.addEventListener('keypress', function(e){
    const now = Date.now();
    const focused = document.activeElement;
    
    // Skip if user is typing in an input/textarea (except testCardInput, which we can auto-fill)
    if(focused && (focused.tagName === 'INPUT' || focused.tagName === 'TEXTAREA')){
      if(focused.id === 'testCardInput'){
        // Allow card reader to fill this field
      } else {
        return; // User is typing in another field
      }
    }
    
    // Reset buffer if too much time passed (human typing vs card reader)
    if(now - lastKeyTime > READER_TIMEOUT){
      cardBuffer = '';
    }
    lastKeyTime = now;
    
    // Enter key = card read complete
    if(e.key === 'Enter' && cardBuffer.length >= CARD_MIN_LENGTH){
      e.preventDefault();
      processPhysicalCard(cardBuffer.trim());
      cardBuffer = '';
      return;
    }
    
    // Accumulate alphanumeric characters (some cards have letters)
    if(/[\dA-Za-z]/.test(e.key)){
      cardBuffer += e.key;
    }
  });

function normalizeEvent(e){
  const ne = { ...e };
  ne.time = ne.time || new Date().toLocaleTimeString();
  ne.point = ne.point || ne.event_point_id || ne.point_id || ne.door_id || '';
  // prefer explicit description over generic event/code
  ne.description = ne.description || ne.event_description || ne.event || (Array.isArray(ne.descriptions)? ne.descriptions.join(', ') : '') || '';
  ne.card_no = ne.card_no || ne.card || ne.card_number || '';
  ne.name = ne.name || ne.employee_name || ne.emp_name || '-';
  ne.cardholder = ne.cardholder || ne.employee_fullname || ne.employee_name || ne.emp_name || '';
  ne.status = ne.status || ne.status_text || '';
  ne.verify_mode = ne.verify_mode || ne.verify || '';
  return ne;
}
function addEvent(obj){
  const e = normalizeEvent(obj);
  const body=document.getElementById('events');
  const tr=document.createElement('tr');
  const alarm = (e.status||'').toUpperCase()==='ALARM' || /(alarm)/i.test(e.description||'');
  
  // If name missing, fill with door name by event point id
  let displayName = (e.name||'').trim();
  try{
    const map = window.__doorMap || {};
    if((!displayName || displayName==='-') && e.point && map[e.point]){
      displayName = map[e.point].name || '-';
    }
    // Fallback: use doorFilter option text that matches the point id
    if((!displayName || displayName==='-') && e.point){
      const sel = document.getElementById('doorFilter');
      if(sel){
        const opt = Array.from(sel.options||[]).find(o=>String(o.value)===String(e.point));
        if(opt){
          // Strip device prefix and id suffix
          const txt = String(opt.textContent||'').replace(/^[^:]+:\s*/,'').replace(/\s*\(id=\d+\)$/, '');
          if(txt){ displayName = txt; }
        }
      }
    }
  }catch{}
  
  // Default background: red for alarms, amber for unknown cards
  let rowBackground = '';
  if(alarm){
    rowBackground = '#3b1e1e'; // Red for alarms
  } else if(e.card_no){
    rowBackground = '#2d3d1e'; // Amber/yellowish for unknown cards initially
  }
  tr.style.background = rowBackground;
  tr.dataset.cardNo = e.card_no || '';
  
  // Create cells - action button will be in the last column (Acțiune)
  const cells = [
    e.time,
    e.point||'',
    e.description||'',
    e.card_no||'',
    displayName || '-',
    (e.cardholder||'') || '-',
    e.status||'',
    e.verify_mode||'',
    '' // Empty for action column - will be filled below
  ].map((val, idx) => {
    // Last column (index 8) is the action button column
    if(idx === 8 && e.card_no){
      return `<td><button data-action-btn style="padding:2px 6px;background:#15803d;color:#fff;border:none;border-radius:3px;cursor:pointer;font-size:11px;font-weight:500;">+Adaugă Card</button></td>`;
    }
    return `<td>${val}</td>`;
  }).join('');
  
  tr.innerHTML = cells;
  
  // Add default click handler
  if(e.card_no && e.card_no.trim()){
    const btn = tr.querySelector('[data-action-btn]');
    if(btn){
      btn.onclick = () => openAddCardModal(e.card_no);
    }
    // Check if card belongs to employee (async, non-blocking)
    checkAndUpdateCardRow(tr, e.card_no);
  }
  
  body.insertBefore(tr, body.firstChild);
  while(body.rows.length>500){ body.deleteRow(500); }
}

// Async function to check card ownership and update row styling
function checkAndUpdateCardRow(tr, cardNumber){
  console.log('[Card Check] Checking card:', cardNumber);
  const url = `/agent/api/check-card-owner/?card_number=${encodeURIComponent(cardNumber)}`;
  console.log('[Card Check] Fetching:', url);
  
  fetch(url)
    .then(r=>{
      console.log('[Card Check] Response status:', r.status);
      return r.json();
    })
    .then(data=>{
      console.log('[Card Check] Response data:', data);
      if(data.exists){
        console.log('[Card Check] ✅ Card found! Employee:', data.employee_name, 'ID:', data.employee_id);
        // Card belongs to employee - update row
        tr.style.background = '#1a3d2a'; // Green for employee cards
        tr.dataset.employeeId = data.employee_id;
        tr.dataset.employeeName = data.employee_name;
        // Update button
        const btn = tr.querySelector('[data-action-btn]');
        if(btn){
          console.log('[Card Check] Updating button text and handler...');
          btn.textContent = '✏️ Modificare';
          btn.style.background = '#0960a8';
          btn.onclick = (e) => {
            e.preventDefault();
            console.log('[Card Check] Opening edit modal for employee', data.employee_id);
            openEditCardModal(data.employee_id, cardNumber);
          };
        } else {
          console.warn('[Card Check] ⚠️ Button not found in row!');
        }
      } else {
        console.log('[Card Check] ❌ Card NOT found in database');
      }
    })
    .catch(err=>{
      console.error('[Card Check] ❌ Error:', err);
    });
}

function wsConnect(){
  const ws=new WebSocket((location.protocol==='https:'?'wss':'ws')+'://'+location.host+'/ws/monitor/');
  window.__wsStatus = 'connecting…'; updateDiagnostics();
  ws.onmessage=(e)=>{
    let data; try{data=JSON.parse(e.data);}catch(err){return;}
    if(data.type==='device.status'||data.type==='device.online'){
      devs[data.device_id]=Object.assign(devs[data.device_id]||{id:data.device_id},{online:data.online!==false,sn:data.serial||data.sn,door_state:data.door_state||'CLOSED'});
      renderDevices();
      window.__wsStatus = 'online'; updateDiagnostics();
    } else if(data.type==='rtlog.batch'){
      addEvent({ point:data.point_id||'', description:`RTLOG x${data.lines?.length||0}` });
    } else if(data.type==='event.batch'){
      if(data.descriptions && data.codes){
        data.descriptions.forEach((desc,i)=>{ addEvent({ point:data.point_id||'', description:desc, code:data.codes[i]}); });
      } else { addEvent({ point:data.point_id||'', description:`Events x${data.count||0}`}); }
    } else if(data.type==='legacy.event'){
      addEvent({
        point: data.point_id||data.point||'',
        description: data.description||data.event||'',
        card_no: data.card_no||data.card||'',
        name: data.name||data.employee_name||'-'
      });
    } else if(/^door\./.test(data.type)){
      const ds = devs[data.device_id] || {id:data.device_id};
      let newState = null;
      if(data.type==='door.open'){ ds.door_state='OPEN'; newState='OPEN'; }
      else if(data.type==='door.close'){ ds.door_state='CLOSED'; newState='CLOSED'; }
      else if(data.type==='door.normal_open'){ ds.door_state='NORMAL_OPEN'; newState='UNLOCKED'; }
      else if(data.type==='door.cancel_alarm'){ ds.door_state='ALARM_CLEARED'; newState='CLOSED'; }
      // Update door state in __doorsCache so icons change color
      if(newState && data.door_id && window.__doorsCache){
        const doorIdx = window.__doorsCache.findIndex(d => d.id === data.door_id);
        if(doorIdx >= 0){
          window.__doorsCache[doorIdx].state = newState;
          renderDevices(); // Re-render to update colors
        }
      }
      devs[data.device_id]=ds; renderDevices();
      try {
        const pid = (data.point_id||data.door_id||'');
        const map = window.__doorMap || {};
        const nm = (pid && map[pid] && map[pid].name) ? map[pid].name : '-';
          addEvent({ description:data.type, point: pid, name: nm });
      } catch { addEvent({ description:data.type, point: (data.point_id||data.door_id||'') }); }
    } else { addEvent(data); }
  };
}
function doorAction(id,door,action){
  // Convert door to number for matching in __doorsCache
  const doorId = parseInt(door) || door;
  console.log('[Door Action] Processing:', action, 'door:', doorId, 'device:', id);
  // Map UI actions to backend endpoints
  const endpointAction = (action==='unlock') ? 'normal-open' : (action==='lock' ? 'close' : action);
  // Enforce local lock semantics: when locked, block 'open' and 'normal-open'
  window.__doorLockMap = window.__doorLockMap || {};
  if(action!=='unlock' && (action==='open' || action==='normal-open') && window.__doorLockMap[doorId]){
    const map = window.__doorMap || {}; const nm = (doorId && map[doorId] && map[doorId].name) || '-';
    addEvent({ description:`door.${action} BLOCKED (LOCK active) (id=${doorId}${nm?' name='+nm:''})`, status:'ERR', verify_mode:'CLIENT', point: doorId, name: nm });
    return;
  }
  
  // UPDATE CACHE IMMEDIATELY for instant UI feedback
  console.log('[Door Action] IMMEDIATE UPDATE: Changing UI...');
  if(window.__doorsCache){
    const doorIdx = window.__doorsCache.findIndex(d => d.id === doorId || String(d.id) === String(doorId));
    if(doorIdx >= 0){
      let newState = null;
      if(action==='open'){ newState = 'OPEN'; }
      else if(action==='close'){ newState = 'CLOSED'; }
      else if(action==='lock'){ newState = 'LOCKED'; }
      else if(action==='unlock'){ newState = 'UNLOCKED'; }
      else if(action==='normal-open'){ newState = 'OPEN'; }
      if(newState){
        console.log('[Door Action] ✅ IMMEDIATELY setting door state to:', newState);
        window.__doorsCache[doorIdx].state = newState;
        renderDevices(); // RE-RENDER NOW
      }
    }
  }
  
  // ASYNC: Send API request
  fetch(`/agent/api/devices/${id}/doors/${doorId}/${endpointAction}/`).then(r=>r.json()).then(j=>{
    console.log('[Door Action] API Response:', j);
    const map = window.__doorMap || {}; const nm = (doorId && map[doorId] && map[doorId].name) || '-';
    const desc = `door.${action} ${j.ok?'OK':'ERR'} (id=${doorId}${nm?' name='+nm:''})`;
    addEvent({ description: desc, status: j.ok?'OK':'ERR', verify_mode:'API', point: doorId, name: nm });
    // Update local lock state based on action
    if(j.ok){
      if(action==='lock'){ window.__doorLockMap[doorId] = true; }
      if(action==='unlock'){ window.__doorLockMap[doorId] = false; }
    }
  }).catch((err)=>{
    console.error('[Door Action] Fetch error:', err);
    const map = window.__doorMap || {}; const nm = (doorId && map[doorId] && map[doorId].name) || '-';
    const desc = `door.${action} ERR (id=${doorId}${nm?' name='+nm:''})`;
    addEvent({ description: desc, status:'ERR', verify_mode:'API', point: doorId, name: nm });
  });
}
function onDoorIconClick(deviceId, doorId){
  // Toggle open/close based on last known state; log only once via API response
  console.log('[Door Click] Clicked on door:', doorId, 'device:', deviceId);
  try {
    window.__doorLockMap = window.__doorLockMap || {};
    if(window.__doorLockMap[doorId]){
      const map = window.__doorMap || {}; const nm = (doorId && map[doorId] && map[doorId].name) || '-';
      addEvent({ description:`door.open BLOCKED (LOCK active) (id=${doorId}${nm?' name='+nm:''})`, status:'ERR', verify_mode:'CLIENT', point: doorId, name: nm });
      return;
    }
    // Get door state from __doorsCache (not devs, which is for devices)
    let isOpen = false;
    if(window.__doorsCache){
      const door = window.__doorsCache.find(d => d.id === doorId || String(d.id) === String(doorId));
      if(door){
        isOpen = (door.state === 'OPEN' || door.state === 'UNLOCKED');
        console.log('[Door Click] Found door state:', door.state, 'isOpen:', isOpen);
      }
    }
    const act = isOpen ? 'close' : 'open';
    console.log('[Door Click] Toggling action:', act);
    if(deviceId && doorId){ doorAction(deviceId, doorId, act); }
  } catch(e){
    console.error('[Door Click] Error:', e);
    addEvent({ description:`door.open ERR`, status:'ERR', verify_mode:'API', point: doorId });
  }
}

function translateReasons(r){
  if(!r) return '';
  const s = (Array.isArray(r)? r.join(', ') : String(r)).toLowerCase();
  return s
    .replaceAll('no_employee_for_card','nu există angajat pentru acest card')
    .replaceAll('outside_time_segments','în afara intervalelor de timp')
    .replaceAll('door_open_failed','deschiderea ușii a eșuat')
    .replaceAll('unauth','neautorizat')
    .replaceAll('not_allowed','acces nepermis');
}
